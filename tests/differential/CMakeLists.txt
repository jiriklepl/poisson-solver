# Local CPU ABI checks never initialize CUDA. GPU differential tests are an
# explicit opt-in with authenticated reference objects, never fetched implicitly.
enable_language(C)
add_executable(device-api-contract device_contract.c)
target_link_libraries(device-api-contract PRIVATE poisson-solver)
add_test(NAME device-api-contract COMMAND device-api-contract)
add_executable(allocation-contract allocation_contract.cpp)
target_link_libraries(allocation-contract PRIVATE poisson-solver)
add_test(NAME allocation-contract COMMAND allocation-contract)
if(POISSON_SOLVER_BUILD_FORTRAN)
    add_executable(device-fortran-contract fortran_contract.f90 fortran_layout.c)
    target_include_directories(device-fortran-contract PRIVATE ${CMAKE_Fortran_MODULE_DIRECTORY})
    target_link_libraries(device-fortran-contract PRIVATE poisson-solver)
    add_test(NAME device-fortran-contract COMMAND device-fortran-contract)
endif()
set(POISSON_SOLVER_POISFFT_BUILD "" CACHE PATH "Pinned reference PoisFFT build with modules and P/*.o")
set(POISSON_SOLVER_POISFFT_LIBS "" CACHE STRING "Pinned FFTW libraries required by the reference objects")
if(POISSON_SOLVER_POISFFT_BUILD)
    if(NOT POISSON_SOLVER_BUILD_FORTRAN)
        message(FATAL_ERROR "Differential reference requires Fortran binding")
    endif()
    set(reference_objects)
    foreach(name customfftw3 fft main parameters poisfft precisions)
        set(object "${POISSON_SOLVER_POISFFT_BUILD}/P/${name}.o")
        if(NOT EXISTS "${object}")
            message(FATAL_ERROR "Missing pinned PoisFFT reference object ${object}")
        endif()
        list(APPEND reference_objects "${object}")
    endforeach()
    add_executable(poisfft-differential test_poisfft.cu poisfft_reference.f90 ${reference_objects})
    target_include_directories(poisfft-differential PRIVATE ${POISSON_SOLVER_POISFFT_BUILD})
    target_link_libraries(poisfft-differential PRIVATE poisson-solver ${POISSON_SOLVER_POISFFT_LIBS} gomp)
    target_compile_options(poisfft-differential PRIVATE $<$<COMPILE_LANGUAGE:Fortran>:-fopenmp>)
    add_test(NAME poisfft-differential COMMAND poisfft-differential)
    set_tests_properties(poisfft-differential PROPERTIES LABELS "cuda;differential" TIMEOUT 240)
endif()

# The cache contract uses NVRTC/CUDA stubs and never initializes a GPU.
find_package(Threads REQUIRED)
add_executable(kernel-cache-contract kernel_cache_host.cpp
    ../../src/detail/kernel_binary_cache.cpp ../../src/detail/nvrtc_cache.cpp)
target_include_directories(kernel-cache-contract PRIVATE
    ../../include "${PROJECT_BINARY_DIR}/generated" ${CUDAToolkit_INCLUDE_DIRS})
target_link_libraries(kernel-cache-contract PRIVATE OpenSSL::Crypto Threads::Threads ${CMAKE_DL_LIBS})
# The assertions are the test, including in a Release provider build.
target_compile_options(kernel-cache-contract PRIVATE -UNDEBUG -Wall -Wextra)
add_test(NAME kernel-cache-contract COMMAND kernel-cache-contract)
set_tests_properties(kernel-cache-contract PROPERTIES LABELS "host;cache" TIMEOUT 30)
add_test(NAME kernel-cache-unknown-runtime COMMAND kernel-cache-contract --unknown-runtime)
set_tests_properties(kernel-cache-unknown-runtime PROPERTIES LABELS "host;cache" TIMEOUT 30)
